💡 방법 1: 사분위수(IQR)를 활용한 이상치 탐지 로직 (실무 표준)
import pandas as pd
df = pd.read_csv('thickness_data.csv')

# 1. 1사분위(25%)와 3사분위(75%) 값 구하기
Q1 = df['Thickness'].quantile(0.25)
Q3 = df['Thickness'].quantile(0.75)

# 2. IQR (사분위수 범위) 계산
IQR = Q3 - Q1

# 3. 정상 범위의 상/하한선 설정 (통상 IQR의 1.5배 적용)
lower_bound = Q1 - 1.5 * IQR
upper_bound = Q3 + 1.5 * IQR

# 4. 상한선을 넘거나 하한선보다 작은 이상치 필터링
outliers = df[(df['Thickness'] < lower_bound) | (df['Thickness'] > upper_bound)]
print(outliers)
